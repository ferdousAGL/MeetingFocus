private fun updateTabLabels() {
val t0 = tv(R.id.tabAll); val t1 = tv(R.id.tabAllowed)
val t2 = tv(R.id.tabOffline); val t3 = tv(R.id.tabFrozen)

t3.text = "❄️ ফ্রিজড (${allApps.count { it.isFrozen }})"

for (i in tabs.indices) {

tabs[i].setTextColor(if (i == currentFilter) Color.WHITE else cl("#94A3B8"))

private fun setFilterTab(index: Int) { currentFilter = index; updateTabLabels(); filterApps() }
private fun killApp(pkg: String) { try { (getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager).killBackgroundProcesses(pkg) } catch (_: Exception) {} }
private fun killAllFrozenApps() { for (pkg in PrefsManager.getFrozenPackages(this)) killApp(pkg) }
private fun startFirewallVpn() {
val i = VpnService.prepare(this)
if (i != null) startActivityForResult(i, 1001) else startService(Intent(this, FocusVpnService::class.java))
}
private fun stopFirewallVpn() { startService(Intent(this, FocusVpnService::class.java).apply { action = FocusVpnService.ACTION_STOP_VPN }) }
override fun onActivityResult(req: Int, res: Int, data: Intent?) {
super.onActivityResult(req, res, data)
if (req == 1001 && res == RESULT_OK) startService(Intent(this, FocusVpnService::class.java))
}
private fun loadInstalledApps() {
val pm = packageManager
val alw = PrefsManager.getAllowedPackages(this); val off = PrefsManager.getOfflinePackages(this); val frz = PrefsManager.getFrozenPackages(this)
val list = pm.getInstalledApplications(0)
.filter { (it.flags and ApplicationInfo.FLAG_SYSTEM) == 0 || alw.contains(it.packageName) || off.contains(it.packageName) || PrefsManager.BROWSERS.contains(it.packageName) }
.filter { it.packageName != packageName }
.map {
val isF = frz.contains(it.packageName); val isO = off.contains(it.packageName) || isF; val isA = alw.contains(it.packageName) && !isO
AppEntry(pm.getApplicationLabel(it).toString(), it.packageName, pm.getApplicationIcon(it), isA, isO, isF)
}
.sortedWith(compareByDescending<AppEntry> { it.isAllowed }.thenByDescending { it.isFrozen }.thenByDescending { it.isOffline }.thenBy { it.appName })
allApps.clear(); allApps.addAll(list)
val rv = findViewById<RecyclerView>(R.id.recyclerApps)
rv.layoutManager = LinearLayoutManager(this)
adapter = AppAdapter(displayedApps); rv.adapter = adapter
updateHeaderAndButtons(); filterApps()
}
private fun filterApps() {
val q = findViewById<EditText>(R.id.edtSearch).text?.toString()?.trim()?.lowercase() ?: ""
displayedApps.clear()
for (a in allApps) {
if (q.isNotEmpty() && !a.appName.lowercase().contains(q) && !a.packageName.lowercase().contains(q)) continue
val m = when (currentFilter) { 1 -> a.isAllowed && !a.isOffline && !a.isFrozen; 2 -> a.isOffline || a.isFrozen; 3 -> a.isFrozen; else -> true }
if (m) displayedApps.add(a)
}
tv(R.id.txtEmptyState).visibility = if (displayedApps.isEmpty()) View.VISIBLE else View.GONE
adapter?.notifyDataSetChanged()
}
