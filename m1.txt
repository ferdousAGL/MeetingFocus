private fun ensureMeetingModeRunning() {
if (!PrefsManager.isMeetingModeActive(this)) {
PrefsManager.setMeetingModeActive(this, true)
suppressSwitch = true; findViewById<SwitchMaterial>(R.id.switchMeetingMode).isChecked = true; suppressSwitch = false
}
killAllFrozenApps(); startFirewallVpn(); updateHeaderAndButtons()
}
private fun toggleAutoSetMode() {
val turningOn = !PrefsManager.isAutoSetActive(this)
PrefsManager.setAutoSetActive(this, turningOn)
val banner = tv(R.id.txtLiveBanner)
if (turningOn) {
val newOff = mutableSetOf<String>(); val newFrz = mutableSetOf<String>()
var alwCnt = 0; var frzCnt = 0
for (app in allApps) {
if (app.isAllowed || PrefsManager.BROWSERS.contains(app.packageName)) {
app.isOffline = false; app.isFrozen = false; alwCnt++
} else {
app.isOffline = true; app.isFrozen = true
newOff.add(app.packageName); newFrz.add(app.packageName); frzCnt++
}
}
PrefsManager.setBulkOfflinePackages(this, newOff)
PrefsManager.setBulkFrozenPackages(this, newFrz)
ensureMeetingModeRunning(); filterApps()
banner.text = "⚡ অটো সেট চালু আছে: $alwCnt টি সচল অ্যাপ বাদে বাকি $frzCnt টি অ্যাপ ফ্রিজ ও নেট অফ করা হয়েছে!"
showPopup(
"⚡ অটো সেট চালু করা হয়েছে!",
"• \uD83D\uDFE2 সচল ও অনলাইন আছে ($alwCnt টি অ্যাপ): আপনার সিলেক্ট করা জরুরি অ্যাপ (যেমন WhatsApp, Phone, SMS) এবং ব্রাউজার সচল রাখা হয়েছে, এগুলোতে কল ও মেসেজ আসবে।\n\n• ❄️ একদম অফ করা হয়েছে ($frzCnt টি অ্যাপ): বাকি সব অ্যাপের ব্যাকগ্রাউন্ড প্রসেস, ইন্টারনেট এবং নোটিফিকেশন সম্পূর্ণ বন্ধ (ফ্রিজ + নেট অফ) করে দেওয়া হয়েছে।\n\n\uD83D\uDCA1 টিপস: যেকোনো অ্যাপকে সচল রাখতে চাইলে ওই অ্যাপের কার্ডে '\uD83D\uDFE2 সচল রাখুন' বাটনে চাপ দিন।"
)
} else {
for (app in allApps) { app.isOffline = false; app.isFrozen = false }
PrefsManager.setBulkOfflinePackages(this, emptySet())
PrefsManager.setBulkFrozenPackages(this, emptySet())
PrefsManager.setMeetingModeActive(this, false)
suppressSwitch = true; findViewById<SwitchMaterial>(R.id.switchMeetingMode).isChecked = false; suppressSwitch = false
stopFirewallVpn(); updateHeaderAndButtons(); filterApps()
banner.text = "⚪ অটো সেট বন্ধ করা হয়েছে: সব অ্যাপ আবার স্বাভাবিক অবস্থায় ফিরে গেছে।"
showPopup("⚪ অটো সেট বন্ধ করা হয়েছে", "সব অ্যাপের ফ্রিজ ও নেট অফ আনলক করা হয়েছে। এখন আপনার ফোনের সব অ্যাপ স্বাভাবিকভাবে চলবে।")
}
}

}
